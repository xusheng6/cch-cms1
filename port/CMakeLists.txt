cmake_minimum_required(VERSION 3.16)
project(cch_port C)
find_package(Threads REQUIRED)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
add_library(cch_engine src/position.c src/movegen.c src/search.c src/book.c src/profiles.c)
target_include_directories(cch_engine PUBLIC include)
add_executable(cch src/main.c src/protocol.c)
target_link_libraries(cch PRIVATE cch_engine Threads::Threads)
add_executable(cch-modern src/main_modern.c src/protocol.c)
target_link_libraries(cch-modern PRIVATE cch_engine Threads::Threads)
add_executable(cch_tests tests/test_engine.c)
target_link_libraries(cch_tests PRIVATE cch_engine)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    # The test program intentionally uses assert() as its compact harness;
    # keep those checks active in Release validation builds too.
    target_compile_options(cch_tests PRIVATE -UNDEBUG)
endif()
add_executable(cch_bookdump src/bookdump.c)
target_link_libraries(cch_bookdump PRIVATE cch_engine)
add_executable(cch_random_positions src/random_positions.c)
target_link_libraries(cch_random_positions PRIVATE cch_engine)
enable_testing()
add_test(NAME engine COMMAND cch_tests)
set(CCH_ORIGINAL_BOOK "" CACHE FILEPATH "Optional private OPENING.LIB for the original-book regression")
if(CCH_ORIGINAL_BOOK)
    if(NOT EXISTS "${CCH_ORIGINAL_BOOK}")
        message(FATAL_ERROR "CCH_ORIGINAL_BOOK does not exist: ${CCH_ORIGINAL_BOOK}")
    endif()
    add_test(NAME original_book COMMAND cch_tests "${CCH_ORIGINAL_BOOK}")
endif()
install(TARGETS cch cch-modern cch_bookdump RUNTIME DESTINATION bin)
